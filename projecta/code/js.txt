document.addEventListener('DOMContentLoaded', () => {
  // Set Current Year in Footer
  const currentYearSpan = document.getElementById('currentYear');
  if (currentYearSpan) {
    currentYearSpan.textContent = new Date().getFullYear();
  }

  // Header Scroll Behavior
  const header = document.getElementById('header');
  window.addEventListener('scroll', () => {
    if (window.scrollY > 30) {
      header?.classList.add('scrolled');
    } else {
      header?.classList.remove('scrolled');
    }
  }, { passive: true });

  // Mobile Menu Toggle
  const menuToggle = document.getElementById('menuToggle');
  const mainNav = document.getElementById('mainNav');
  const navList = mainNav?.querySelector('.nav-list');
  const navLinks = document.querySelectorAll('.nav-link');

  if (menuToggle && navList) {
    menuToggle.addEventListener('click', () => {
      const isOpen = navList.classList.toggle('open');
      menuToggle.classList.toggle('active');
      menuToggle.setAttribute('aria-expanded', String(isOpen));
    });

    // Close menu when link is clicked
    navLinks.forEach(link => {
      link.addEventListener('click', () => {
        navList.classList.remove('open');
        menuToggle.classList.remove('active');
        menuToggle.setAttribute('aria-expanded', 'false');
      });
    });

    // Close menu when clicking outside
    document.addEventListener('click', (e) => {
      if (!header?.contains(e.target)) {
        navList.classList.remove('open');
        menuToggle.classList.remove('active');
        menuToggle.setAttribute('aria-expanded', 'false');
      }
    });
  }

  // Active Link Highlighting on Scroll
  const sections = document.querySelectorAll('section[id]');
  window.addEventListener('scroll', () => {
    const scrollPos = window.scrollY + 120;
    sections.forEach(section => {
      const top = section.offsetTop;
      const height = section.offsetHeight;
      const id = section.getAttribute('id');
      const targetLink = document.querySelector(`.nav-link[href="#${id}"]`);

      if (scrollPos >= top && scrollPos < top + height) {
        navLinks.forEach(l => l.classList.remove('active'));
        targetLink?.classList.add('active');
      }
    });
  }, { passive: true });

  // Scroll to Top Button
  const scrollTopBtn = document.getElementById('scrollTopBtn');
  if (scrollTopBtn) {
    window.addEventListener('scroll', () => {
      if (window.scrollY > 400) {
        scrollTopBtn.classList.add('visible');
      } else {
        scrollTopBtn.classList.remove('visible');
      }
    }, { passive: true });

    scrollTopBtn.addEventListener('click', () => {
      window.scrollTo({
        top: 0,
        behavior: 'smooth'
      });
    });
  }

  // Contact Form Handling & Validation
  const contactForm = document.getElementById('contactForm');
  const submitBtn = document.getElementById('submitBtn');
  const successBanner = document.getElementById('successBanner');
  const resetFormBtn = document.getElementById('resetFormBtn');
  const whatsappForwardBtn = document.getElementById('whatsappForwardBtn');

  // Input elements
  const fullNameInput = document.getElementById('fullName');
  const phoneInput = document.getElementById('phone');
  const emailInput = document.getElementById('email');
  const serviceInput = document.getElementById('serviceInterest');
  const otherDeptGroup = document.getElementById('otherDeptGroup');
  const otherDeptInput = document.getElementById('otherDeptInput');
  const messageInput = document.getElementById('message');

  // Helper validation functions
  const validateEmail = (email) => {
    return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email.trim());
  };

  const validatePhone = (phone) => {
    const digits = phone.replace(/\D/g, '');
    return digits.length >= 10;
  };

  const setFieldError = (input, hasError) => {
    const formGroup = input.closest('.form-group');
    if (formGroup) {
      if (hasError) {
        formGroup.classList.add('has-error');
      } else {
        formGroup.classList.remove('has-error');
      }
    }
  };

  // Clear errors when typing
  [fullNameInput, phoneInput, emailInput, messageInput, otherDeptInput].forEach(input => {
    if (!input) return;
    input.addEventListener('input', () => {
      setFieldError(input, false);
    });
  });

  // Dynamic Toggle for 'Other' Department
  if (serviceInput && otherDeptGroup) {
    serviceInput.addEventListener('change', () => {
      setFieldError(serviceInput, false);
      if (serviceInput.value === 'Other') {
        otherDeptGroup.hidden = false;
        otherDeptInput?.focus();
      } else {
        otherDeptGroup.hidden = true;
        if (otherDeptInput) {
          otherDeptInput.value = '';
          setFieldError(otherDeptInput, false);
        }
      }
    });
  }

  if (contactForm) {
    contactForm.addEventListener('submit', (e) => {
      e.preventDefault();

      let isValid = true;
      let firstInvalidInput = null;

      // Validate Student Name
      if (!fullNameInput.value.trim() || fullNameInput.value.trim().length < 2) {
        setFieldError(fullNameInput, true);
        isValid = false;
        if (!firstInvalidInput) firstInvalidInput = fullNameInput;
      } else {
        setFieldError(fullNameInput, false);
      }

      // Validate Mobile Phone
      if (!validatePhone(phoneInput.value)) {
        setFieldError(phoneInput, true);
        isValid = false;
        if (!firstInvalidInput) firstInvalidInput = phoneInput;
      } else {
        setFieldError(phoneInput, false);
      }

      // Validate Email
      if (!validateEmail(emailInput.value)) {
        setFieldError(emailInput, true);
        isValid = false;
        if (!firstInvalidInput) firstInvalidInput = emailInput;
      } else {
        setFieldError(emailInput, false);
      }

      // Validate Department Selection
      if (!serviceInput.value) {
        setFieldError(serviceInput, true);
        isValid = false;
        if (!firstInvalidInput) firstInvalidInput = serviceInput;
      } else {
        setFieldError(serviceInput, false);
      }

      // Validate 'Other' Department when selected
      if (serviceInput.value === 'Other') {
        if (!otherDeptInput || !otherDeptInput.value.trim()) {
          if (otherDeptInput) setFieldError(otherDeptInput, true);
          isValid = false;
          if (!firstInvalidInput) firstInvalidInput = otherDeptInput;
        } else {
          setFieldError(otherDeptInput, false);
        }
      }

      // Validate Project Topic & Message
      if (!messageInput.value.trim() || messageInput.value.trim().length < 10) {
        setFieldError(messageInput, true);
        isValid = false;
        if (!firstInvalidInput) firstInvalidInput = messageInput;
      } else {
        setFieldError(messageInput, false);
      }

      if (!isValid) {
        firstInvalidInput?.focus();
        return;
      }

      // Show Loading State
      submitBtn.classList.add('loading');
      submitBtn.disabled = true;

      // Determine final selected department
      const finalDepartment = serviceInput.value === 'Other'
        ? `Other (${otherDeptInput.value.trim()})`
        : serviceInput.value;

      // Extract cleaned values
      const studentName = fullNameInput.value.trim();
      const studentPhone = phoneInput.value.trim();
      const studentEmail = emailInput.value.trim();
      const projectDetails = messageInput.value.trim();

      // Formulate WhatsApp message text for ProjectA
      const waText = encodeURIComponent(
        `*New College Project Order via ProjectA:*\n\n` +
        `• *Student Name:* ${studentName}\n` +
        `• *Mobile:* ${studentPhone}\n` +
        `• *Email:* ${studentEmail}\n` +
        `• *Department/Branch:* ${finalDepartment}\n` +
        `• *Project Topic & Requirements:* ${projectDetails}`
      );
      const waUrl = `https://wa.me/919714711897?text=${waText}`;

      if (whatsappForwardBtn) {
        whatsappForwardBtn.href = waUrl;
      }

      // Populate order summary card for student confirmation
      const orderSummaryBox = document.getElementById('orderSummaryBox');
      if (orderSummaryBox) {
        orderSummaryBox.innerHTML = `
          <strong style="color: #0066ff; display: block; margin-bottom: 0.4rem;">📋 Order Details Sent:</strong>
          <div><strong>Student:</strong> ${studentName}</div>
          <div><strong>Mobile:</strong> ${studentPhone}</div>
          <div><strong>Email:</strong> ${studentEmail}</div>
          <div><strong>Department:</strong> ${finalDepartment}</div>
          <div><strong>Topic:</strong> ${projectDetails}</div>
        `;
      }

      // 1. Send Order Details to support.projecta@gmail.com via FormSubmit AJAX
      fetch('https://formsubmit.co/ajax/support.projecta@gmail.com', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Accept': 'application/json'
        },
        body: JSON.stringify({
          "Student Name": studentName,
          "WhatsApp Mobile": studentPhone,
          "Email Address": studentEmail,
          "Department": finalDepartment,
          "Project Requirements": projectDetails,
          "_subject": `New Project Order: ${studentName} (${finalDepartment})`,
          "_template": "table",
          "_captcha": "false"
        })
      })
      .then(response => response.json())
      .then(data => {
        console.log('Project order emailed to support.projecta@gmail.com:', data);
      })
      .catch(err => {
        console.error('Email dispatch error (offline/blocked):', err);
      })
      .finally(() => {
        submitBtn.classList.remove('loading');
        submitBtn.disabled = false;

        // 2. Open WhatsApp with pre-filled details directly to 9714711897
        window.open(waUrl, '_blank');

        // 3. Hide form and show success confirmation
        contactForm.style.display = 'none';
        if (successBanner) {
          successBanner.hidden = false;
          successBanner.scrollIntoView({ behavior: 'smooth', block: 'center' });
        }
      });
    });
  }

  // Reset form to send another message
  if (resetFormBtn && contactForm && successBanner) {
    resetFormBtn.addEventListener('click', () => {
      contactForm.reset();
      [fullNameInput, phoneInput, emailInput, serviceInput, otherDeptInput, messageInput].forEach(input => {
        if (input) setFieldError(input, false);
      });
      if (otherDeptGroup) {
        otherDeptGroup.hidden = true;
      }
      if (orderSummaryBox) {
        orderSummaryBox.innerHTML = '';
      }
      successBanner.hidden = true;
      contactForm.style.display = 'flex';
      fullNameInput?.focus();
    });
  }
});
